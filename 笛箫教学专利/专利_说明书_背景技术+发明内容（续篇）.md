### 步骤 J2：构建粉红噪声发生器

**J2.1** 粉红噪声的物理特性：粉红噪声（Pink Noise）的功率谱密度与频率成反比（每倍频程下降约3dB），其频谱特性比白噪声（所有频率等功率）更接近真实自然环境中的背景声音——如风吹树叶、远处的人声、空调运转声。

**J2.2** 采用 Voss-McCartney 算法生成粉红噪声：

```javascript
function generatePinkNoise(durationSeconds) {
    // J2.2.1：创建粉红噪声的音频缓冲区
    let sampleRate = noiseCtx.sampleRate;
    let numSamples = sampleRate * durationSeconds;  // 例如 44100 * 2 = 88200 样本
    let buffer = noiseCtx.createBuffer(1, numSamples, sampleRate);
    let data = buffer.getChannelData(0);  // 获取第0声道（单声道）的 Float32Array 引用
    
    // J2.2.2：Voss-McCartney 算法——6 个独立的随机游走
    let b = [0, 0, 0, 0, 0, 0];  // 6 个游走的当前值
    for (let i = 0; i < numSamples; i++) {
        // J2.2.2.1：对每个游走，以概率 1/2^n 更新其值
        // n=0 → 概率1/1=100% 每次更新 → 高频噪声
        // n=5 → 概率1/32≈3% 更新   → 低频漂移
        for (let n = 0; n < 6; n++) {
            if (Math.random() < Math.pow(2, -n)) {
                b[n] = Math.random() * 2 - 1;  // 生成 [-1, 1] 区间的随机值
            }
        }
        // J2.2.2.2：将 6 个游走的当前值求和并缩放到适当幅度
        // 0.16 = 1/(6+0.5) 的近似值，确保输出幅度不超过[-1, 1]
        data[i] = (b[0] + b[1] + b[2] + b[3] + b[4] + b[5]) * 0.16;
    }
    
    return buffer;
}
```

**J2.3** 组装并启动噪音播放器：

```javascript
let noiseSource = null;
let noiseGainNode = null;
let noiseCtx = null;

function startNoiseGenerator(scenarioType) {
    // J2.3.1：获取当前场景配置中的噪音强度参数
    let cfg = scenarioConfig.scenarios[scenarioType];
    let targetDB = cfg.noise_db;  // 目标噪音强度（dB）
    
    // J2.3.2：如果噪音强度为 0（入门模式），直接返回，不启动噪音
    if (targetDB <= 0) {
        console.log('当前场景无需背景噪音（噪音强度=0dB）');
        return;
    }
    
    // J2.3.3：创建独立的 AudioContext（与诊断管线的 audioCtx 分开）
    noiseCtx = new (window.AudioContext || window.webkitAudioContext)();
    
    // J2.3.4：生成 2 秒粉红噪声缓冲区
    noiseBuffer = generatePinkNoise(2);
    
    // J2.3.5：创建循环播放的噪音源节点
    noiseSource = noiseCtx.createBufferSource();
    noiseSource.buffer = noiseBuffer;
    noiseSource.loop = true;  // 循环播放——噪音持续存在直到手动停止
    
    // J2.3.6：创建增益节点并设置目标音量
    // 增益值换算公式：gain = 10^((dB - 100) / 20)
    // 例如：targetDB=50 → gain = 10^((50-100)/20) = 10^(-2.5) ≈ 0.00316
    noiseGainNode = noiseCtx.createGain();
    let gainValue = Math.pow(10, (targetDB - 100) / 20);
    // 使用渐变式增益变化（2秒内平滑过渡），避免突然的噪音爆音
    noiseGainNode.gain.setValueAtTime(0.0001, noiseCtx.currentTime);  // 当前：几乎静音
    noiseGainNode.gain.linearRampToValueAtTime(gainValue, noiseCtx.currentTime + 2); // 2秒后：目标音量
    
    // J2.3.7：连接节点链：source → gain → destination
    noiseSource.connect(noiseGainNode);
    noiseGainNode.connect(noiseCtx.destination);
    
    // J2.3.8：启动播放
    noiseSource.start(0);
    console.log('粉红噪声音频已启动，目标音量：' + targetDB + 'dB');
}

function stopNoiseGenerator() {
    if (noiseSource) {
        noiseSource.stop(0);       // 立即停止
        noiseSource.disconnect();  // 断开连接，释放资源
        noiseSource = null;
    }
    if (noiseCtx) {
        noiseCtx.close();          // 关闭 AudioContext，彻底释放系统音频资源
        noiseCtx = null;
    }
}
```

### 步骤 J3：构建合奏延迟模拟器

**J3.1** 延迟模拟的核心逻辑：在合奏排练场景中，系统需要一个可调变量来控制"合奏伙伴声部"的播放时间偏移。通过逐步减小这个偏移量（从50ms降至0ms），学习者逐渐适应完美的合奏对齐。

**J3.2** 实现代码：

```javascript
let delayNode = null;
let currentDelay = 0;  // 当前合奏延迟量（ms）

function setupEnsembleDelay(initialDelayMs) {
    // J3.2.1：在主 AudioContext 中创建 DelayNode
    //         注意：此处使用诊断管线的 audioCtx（非噪音的 noiseCtx）
    delayNode = audioCtx.createDelay(1.0);  // 最大延迟时间 = 1秒
    
    // J3.2.2：设置初始延迟时间（单位：秒）
    currentDelay = initialDelayMs;  // 存储毫秒值（方便后续梯度调整）
    delayNode.delayTime.setValueAtTime(initialDelayMs / 1000, audioCtx.currentTime);
    // 解释：将毫秒转换为秒（50ms → 0.05s），DelayNode 的 delayTime 属性单位为秒
    
    // J3.2.3：将合奏伴奏音频源接入 DelayNode
    // ensembleSource 是预录制的合奏伴奏轨的 AudioBufferSourceNode
    ensembleSource.connect(delayNode);
    delayNode.connect(audioCtx.destination);
    // 解释：伴奏音频 → DelayNode（延迟 currentDelay 秒）→ 扬声器输出
    //       学习者的实时演奏是直接听到的，合奏声部是延迟后听到的
    //       学习者需要适应"自己先吹、合奏声部后到"的体验
    //       当 currentDelay = 0 时，两个声部完美对齐
}

function decreaseDelayByStep(stepMs) {
    // J3.2.4：梯度式减小延迟
    if (currentDelay <= 0) return;  // 已经是 0ms，不能更小
    
    currentDelay = Math.max(0, currentDelay - stepMs);
    // 例如：currentDelay=50, stepMs=10 → newDelay=40ms
    //       currentDelay=10, stepMs=10 → newDelay=0ms（不低于0）
    
    // 使用线性渐变（Ramp）平滑过渡，避免延迟跳变引起的音频爆音（click）
    delayNode.delayTime.linearRampToValueAtTime(
        currentDelay / 1000, 
        audioCtx.currentTime + 0.1  // 在100ms内平滑过渡到新延迟值
    );
    
    console.log('合奏延迟已调整为 ' + currentDelay + 'ms');
}

// J3.2.5：自动梯度训练循环
// 每完成一次合奏即触发：延迟减少一个步进
document.addEventListener('ensemble:run_complete', function() {
    decreaseDelayByStep(10);  // 每次减少 10ms
    if (currentDelay <= 0) {
        document.dispatchEvent(new CustomEvent('ensemble:perfect_sync', {
            detail: { message: '🎉 完美对齐！你已适应0ms延迟的合奏。' }
        }));
    }
});
```

### 步骤 J4：建立 BLE 触觉反馈通信

**J4.1** BLE 连接流程（5步握手）：

```
步骤 S1：浏览器扫描附近的 BLE 设备 → 过滤出服务 UUID 匹配的手环
步骤 S2：用户从弹出列表中选择手环 → 浏览器发起 GATT 连接
步骤 S3：连接成功后 → 获取指定的 GATT Service
步骤 S4：在 Service 中 → 获取指定的 GATT Characteristic（振动指令写入特征值）
步骤 S5：向 Characteristic 写入振动指令字节 → 手环执行振动
```

**J4.2** 完整代码实现：

```javascript
// 注意：以下 UUID 为示例值，实际值需与手环硬件固件中的 Service/Characteristic UUID 匹配
const HAPTIC_SERVICE_UUID = '0000xxxx-0000-1000-8000-00805f9b34fb';
const HAPTIC_CHAR_UUID     = '0000yyyy-0000-1000-8000-00805f9b34fb';

let btDevice = null;
let btChar = null;

async function connectHapticBand() {
    try {
        // J4.2.1 步骤 S1：请求用户选择 BLE 手环设备
        btDevice = await navigator.bluetooth.requestDevice({
            filters: [{ services: [HAPTIC_SERVICE_UUID] }],
            // 解释：filters 数组指定了"只显示广播此 Service UUID 的 BLE 设备"
            //       如果手环没有广播 Service UUID（大多数 BLE 外设不广播 Service UUID），
            //       使用 acceptAllDevices: true 替代 filters
            optionalServices: [HAPTIC_SERVICE_UUID]
        });
        console.log('已选择设备：' + btDevice.name);
        
        // J4.2.2 步骤 S2：建立 GATT 连接
        let server = await btDevice.gatt.connect();
        console.log('GATT 连接成功');
        
        // J4.2.3 步骤 S3：获取 Service
        let service = await server.getPrimaryService(HAPTIC_SERVICE_UUID);
        
        // J4.2.4 步骤 S4：获取 Characteristic
        btChar = await service.getCharacteristic(HAPTIC_CHAR_UUID);
        console.log('振动特征值获取成功');
        
        // J4.2.5 步骤 S5：发送初始测试振动（一条短振，确认通信通路正常）
        await sendHapticCommand({
            mode: 0x01,   // 短振模式（Short Vibration）
            frequency: 8, // 8 Hz 振动频率
            duration: 5   // 持续时间 = duration * 10ms = 50ms
        });
        console.log('手环通信测试完成——连接成功');
        
    } catch (err) {
        // J4.2.6 分支：连接失败处理
        if (err.name === 'NotFoundError') {
            console.warn('未找到手环设备，境模块将以无触觉反馈模式运行');
            // 系统仍可正常运行，只是缺少触觉反馈维度
            return false;
        }
        if (err.name === 'SecurityError') {
            console.warn('用户取消了 BLE 配对，境模块将以无触觉反馈模式运行');
            return false;
        }
        throw err;  // 其他错误向上抛出
    }
    return true;
}

// J4.3：振动指令编码协议
// 每个振动指令由 3 个字节组成（不包括 GATT 协议头部）：
// 字节 1（byte 0）：振动模式（0x01=短振 0x02=节拍器 0x03=情感提示 0x04=偏差预警）
// 字节 2（byte 1）：振动频率，单位 Hz，有效范围 2-15
// 字节 3（byte 2）：持续时间/10，单位 ms/10（即 byte=5 表示 50ms，取值范围 1-255）

async function sendHapticCommand(cmd) {
    if (!btChar) {
        console.warn('手环未连接，跳过触觉指令发送');
        return;
    }
    
    // J4.3.1：按协议编码指令字节
    let bytes = new Uint8Array([cmd.mode, cmd.frequency, cmd.duration]);
    
    // J4.3.2：通过 GATT Write 发送指令字节
    try {
        await btChar.writeValue(bytes);
        // writeValue() 返回 Promise：
        // - resolve → 手环收到指令并应答
        // - reject  → 连接断开或写入超时（手动重连）
    } catch (writeErr) {
        console.error('触觉指令发送失败：' + writeErr.message);
        btChar = null;
        // 触发重连
        setTimeout(connectHapticBand, 3000);  // 3 秒后自动尝试重连
    }
}
```

## 装配步骤

### 步骤 C1：MFCC 特征提取（17步详细流程）

**C1.1** MFCC（Mel-Frequency Cepstral Coefficients，梅尔频率倒谱系数）是语音和音乐信号处理领域最常用的声学特征之一。它模拟了人耳对音高的非线性感知特性——人耳对低频音高的分辨力远高于高频——通过对频率轴进行梅尔尺度变换来纳入这一特性。

**C1.2** 从线性频率 f（Hz）到梅尔频率 mel(f) 的转换公式：

```
mel(f) = 2595 * log10(1 + f / 700)
```

其中 f 为线性频率（Hz）。例如：f=440Hz → mel(440)=2595*log10(1+0.629)≈549 mel；f=2000Hz → mel(2000)=2595*log10(1+2.857)≈1478 mel。

**C1.3** MFCC 提取的 17 步流程（伪代码）：

```
输入：audioBuffer（即兴演奏音频）
参数：sampleRate = audioBuffer.sampleRate（通常 44100 Hz）
      frameDuration = 0.150 秒（150ms）
      hopRatio = 0.5（50% 帧重叠）
      numMelFilters = 26（梅尔滤波器数量）
      numMFCC = 24（输出的 MFCC 系数个数）

S1:   frameSize   = sampleRate * frameDuration           // 每帧样本数 ≈ 6615
S2:   hopSize     = frameSize * hopRatio                // 帧移 ≈ 3307 样本
S3:   numFrames   = floor((totalSamples - frameSize) / hopSize) + 1  // 总帧数
S4:   audioData   = audioBuffer.getChannelData(0)       // 获取第0声道（单声道）样本
S5:   mfccMatrix  = new Array(numFrames)                 // 初始化 MFCC 矩阵（行=帧，列=系数）

S6:   FOR frameIndex = 0 TO numFrames - 1:
S7:       startSample = frameIndex * hopSize             // 当前帧的起始样本索引
S8:       frame = audioData.slice(startSample, startSample + frameSize)  // 截取当前帧

S9:       // ---- 加汉明窗（Hamming Window） ----
S10:      windowedFrame = new Float32Array(frameSize)
S11:      FOR i = 0 TO frameSize - 1:
S12:          windowedFrame[i] = frame[i] * (0.54 - 0.46 * cos(2 * PI * i / (frameSize - 1)))
            // 汉明窗函数：w[i] = 0.54 - 0.46 * cos(2πi/(N-1))
            // 作用：减少帧边缘的频谱泄漏（Spectral Leakage）

S13:      // ---- 快速傅里叶变换（FFT） ----
          // 补零到 2 的整数次幂（如 8192）
S14:      fftSize = nextPowerOfTwo(frameSize)            // 找到 ≥ frameSize 的最小2的幂
S15:      spectrum = FFT(windowedFrame, fftSize)         // 复数频谱
S16:      powerSpectrum = |spectrum|²                    // 功率谱 = 频谱幅度的平方

S17:      // ---- 梅尔滤波器组 ----
S18:      melEnergies = new Float32Array(numMelFilters)   // 26 维 Mel 能量向量

S19:      // ---- 构建 26 个三角滤波器 ----
          // 每个滤波器 i 覆盖一段梅尔频率区间 [melLow, melHigh]
          // 该区间在线性频率轴上对应 [fLow, fHigh]
          // 滤波器的频率响应为三角形状：在中心频率处=1，在边界频率处=0
S20:      FOR filterIndex = 0 TO numMelFilters - 1:
S21:          melEnergies[filterIndex] = 0
S22:          FOR each frequency bin with energy in powerSpectrum:
                  // 计算该 bin 对第 filterIndex 个滤波器的贡献
                  IF bin_freq is within [fLow, fHigh]:
                      weight = triangularWeight(bin_freq, fLow, fCenter, fHigh)
                      melEnergies[filterIndex] += weight * powerSpectrum[bin]

S23:      // ---- 取对数 ----
S24:      logMelEnergies = new Float32Array(numMelFilters)
S25:      FOR i = 0 TO numMelFilters - 1:
S26:          logMelEnergies[i] = log(max(melEnergies[i], 1e-10))
            // 加 1e-10 防止 log(0) = -Infinity

S27:      // ---- 离散余弦变换（DCT） ----
          // DCT 将 26 维 log-mel 能量向量转换为 cepstral 域
          // DCT 系数 C[k] = Σ logMelEnergies[n] * cos(π * k * (n + 0.5) / N)
          // 取前 24 个 DCT 系数作为 MFCC
S28:      mfccVector = new Float32Array(numMFCC)           // 24 维 MFCC 向量
S29:      FOR k = 0 TO numMFCC - 1:
S30:          sum = 0
S31:          FOR n = 0 TO numMelFilters - 1:
S32:              sum += logMelEnergies[n] * cos(PI * k * (n + 0.5) / numMelFilters)
S33:          mfccVector[k] = sum

S34:      mfccMatrix[frameIndex] = mfccVector             // 存入 MFCC 矩阵

// 输出：mfccMatrix = [24维向量] × numFrames 帧
```

**C1.4** 关键参数选择说明：

| 参数 | 选择值 | 选择理由 |
|------|--------|---------|
| frameDuration=150ms | 150ms | 笛箫音符最小时长约 100ms，150ms帧可确保每帧至少包含一个完整音符，同时提供足够的时间分辨率 |
| hopRatio=0.5 | 50% | 半重叠在计算量与时间平滑性之间的标准折中选择 |
| numMelFilters=26 | 26 | 业界标准（如 HTK、Librosa 默认值）——覆盖 0-8000Hz 的频率范围（笛箫音域的 4-5 倍余量） |
| numMFCC=24 | 24 | 保留低频倒谱系数（携带音色/音质信息），丢弃高频系数（携带激励源/噪声信息） |

### 步骤 C2：PCA 降维（24维 → 8维）

**C2.1** PCA（主成分分析）的目的：24维 MFCC 向量中有大量冗余信息（相邻 MFCC 系数高度相关）。PCA 将 24维降为 8维核心特征向量，在保留 ≥90% 方差信息的前提下减少计算量和存储量。

**C2.2** PCA 计算流程：

```
输入：mfccMatrix  [numFrames × 24]

S1： // ---- 数据中心化 ----
S2： meanVector[24] = 逐列求均值(mfccMatrix)
S3： centeredMatrix = mfccMatrix - meanVector  // 每行减去均值向量

S4： // ---- 计算协方差矩阵 ----
S5： covarianceMatrix[24][24] = (centeredMatrix^T * centeredMatrix) / (numFrames - 1)
     // 解释：协方差矩阵的对角线元素 cov[i][i] = 第 i 维 MFCC 系数的方差
     //       非对角线元素 cov[i][j] = 第 i 维和第 j 维 MFCC 系数的协方差
     //       协方差 > 0 → 两维正相关；协方差 < 0 → 两维负相关

S6： // ---- 特征值分解 ----
S7： eigenvalues[24], eigenvectors[24][24] = EIG(covarianceMatrix)
     // 解释：eigenvectors 的每一列是一个特征向量（主成分方向）
     //       eigenvalues 对应每个主成分方向上的方差大小

S8： // ---- 按特征值降序排列，取前 8 个主成分 ----
S9： sortedIndices = argsort(eigenvalues, descending=true)
S10: top8Eigenvectors = eigenvectors[:, sortedIndices[0:8]]  // shape:[24 × 8]

S11: // ---- 投影 ----
S12: pcaMatrix = centeredMatrix * top8Eigenvectors  // shape:[numFrames × 8]

// 输出：pcaMatrix [numFrames × 8]
```

### 步骤 C3：余弦相似度匹配与风格评估

**C3.1** 余弦相似度公式：

```
cos(θ) = (v·u) / (||v|| × ||u||)
         = Σ(v[i] * u[i]) / ( sqrt(Σ v[i]²) * sqrt(Σ u[i]²) )
```

其中 v 为学习者的即兴演奏特征向量（8维 PCA 均值），u 为风格模板的特征向量。cos(θ) 取值范围 [-1, 1]：1→完全同向（风格完全相同），0→正交（无关联），-1→反向（极罕见，通常重置为 0）。

**C3.2** 风格模板匹配与评分函数：

```javascript
function evaluateCreativity(pcaMatrix, styleTemplates) {
    // C3.2.1：计算学习者在所有帧上的平均特征向量（8维）
    let avgFeatureVector = new Float32Array(8);
    for (let f = 0; f < pcaMatrix.length; f++) {
        for (let d = 0; d < 8; d++) {
            avgFeatureVector[d] += pcaMatrix[f][d];
        }
    }
    for (let d = 0; d < 8; d++) {
        avgFeatureVector[d] /= pcaMatrix.length;
    }
    
    // C3.2.2：与所有风格模板逐一计算余弦相似度
    let similarities = [];  // [{templateId, templateName, score}]
    for (let tpl of styleTemplates.templates) {
        let dotProduct = 0;
        let normV = 0, normU = 0;
        for (let d = 0; d < 8; d++) {
            dotProduct += avgFeatureVector[d] * tpl.featureVector[d];
            normV += avgFeatureVector[d] * avgFeatureVector[d];
            normU += tpl.featureVector[d] * tpl.featureVector[d];
        }
        let cosine = dotProduct / (Math.sqrt(normV) * Math.sqrt(normU));
        // 如果余弦值 < 0，重置为 0（在实际音乐风格匹配中极少出现负相关）
        if (cosine < 0) cosine = 0;
        
        similarities.push({
            templateId: tpl.id,
            templateName: tpl.name,
            score: cosine
        });
    }
    
    // C3.2.3：按相似度得分降序排列
    similarities.sort(function(a, b) { return b.score - a.score; });
    
    // C3.2.4：计算三联创造性评分
    let styleFit = similarities[0].score;                    // 风格契合度
    let styleDistinction = similarities[0].score - similarities[1].score;  // 风格辨识度
    // 风格辨识度 ≥ 0.2 → 有明显风格偏向；< 0.1 → 风格模糊
    
    // C3.2.5：即兴复杂度（创新特征占比）
    // 计算学习者的特征向量与所有模板正包络的覆盖比例
    // 超出包络 → 即学习者独有的、不属于任何已有模板的特征 → 创新
    let inEnvelope = 0, outEnvelope = 0;
    for (let d = 0; d < 8; d++) {
        let dMin = Infinity, dMax = -Infinity;
        for (let tpl of styleTemplates.templates) {
            if (tpl.featureVector[d] < dMin) dMin = tpl.featureVector[d];
            if (tpl.featureVector[d] > dMax) dMax = tpl.featureVector[d];
        }
        if (avgFeatureVector[d] >= dMin && avgFeatureVector[d] <= dMax) {
            inEnvelope++;
        } else {
            outEnvelope++;
        }
    }
    let complexity = outEnvelope / 8;  // 创新特征占比
    
    // C3.2.6：加权综合评分
    let overallScore = (0.4 * styleFit + 0.3 * styleDistinction + 0.3 * complexity) * 100;
    // 映射到 0-100 分
    
    return {
        overallScore: Math.round(overallScore),
        styleFit: (styleFit * 100).toFixed(1) + '%',
        styleDistinction: (styleDistinction * 100).toFixed(1) + '%',
        complexity: (complexity * 100).toFixed(1) + '%',
        topMatch: similarities[0].templateName,
        topConfidence: (similarities[0].score * 100).toFixed(1) + '%',
        secondMatch: similarities[1].templateName,
        allMatches: similarities.slice(0, 5)  // 返回前5个最佳匹配
    };
}
```

